# Fase 1.9 - Submenu Del Area Personal - Plan De Implementacion

> **For agentic workers:** REQUIRED SUB-SKILL: Use `writing-plans` to create the detailed plan for
> each phase and `executing-plans` to implement it task-by-task. Steps use checkbox (`- [ ]`) syntax
> for tracking.

**Goal:** Convertir `/perfil` en un area personal con marco comun y submenu responsive de cuatro
destinos independientes: `/perfil/datos`, `/perfil/seguridad`, `/perfil/actividades` y
`/perfil/certificados`.

**Architecture:** Un `PersonalAreaLayout` de la feature `auth` actua como layout de ruta: posee el
`h1` del area, el submenu y el `Outlet` de la seccion activa. La visibilidad de la navegacion se
resuelve en JavaScript con `useMediaQuery` de Chakra, no con una media query de CSS, para que el
estado expandido sea observable en pruebas y no dependa de como jsdom evalua reglas de medios. El
contenido de cada seccion reutiliza los formularios existentes sin modifications de dominio:
`ProfileView` y `ChangePasswordForm` no cambian.

**Tech Stack:** React 19, TypeScript 6, Chakra UI 3, React Router 8, Vitest 5, Testing Library,
Storybook 10, Playwright.

---

## Alcance Y Decisiones

- **Sin cambio de contrato:** la fase no introduce, modifica ni elimina peticiones HTTP, por lo que
  no se consulta OpenAPI y no aplica `pnpm run api:mocks-check`.
- **`/perfil` y `/cambiar-contrasena` se conservan** como accesos de entrada. Ambos redirigen con
  `replace` a su ruta canonica, de modo que no quedan dos rutas equivalentes y el enlace compartido
  ni el marcador deayer siguen funcionando.
- **`AppMenu` y `resolveAuthLandingPath` no cambian.** "Mi perfil" sigue apuntando a `/perfil` y el
  destino tras login sigue siendo `/perfil`; el indice del layout resuelve a la seccion de datos.
- **`h1` en el layout, `h2` en la seccion.** `ProfileForm` y `ChangePasswordForm` ya presentan su
  propio `h2`, asi que las paginas de datos y seguridad no agregan un encabezado propio: la
  jerarquia queda `h1` del area, `h2` del formulario o del estado informativo.
- **Un unico `nav`.** La navegacion se renderiza una sola vez; en movil se oculta con `display` y se
  revela con un boton explicito. No hay dos conjuntos de enlaces ni dos landmarks de navegacion.
- **Sin cambios en `ProfileForm` ni `ChangePasswordForm`.** La seguridad ya no depende del catalogo
  institucional porque vive en su propia ruta, lo que hace imposible que un fallo del catalogo la
  esconda.
- **Sin story movil en el catalogo visual.** `playwright.storybook.config.ts` fija el viewport en
  1280x720, y anadir `@storybook/addon-viewport` para una sola story es una dependencia nueva. La
  rama movil se cubre con pruebas de jsdom y con una comprobacion puntual de Playwright en la Task 9.
  La infraestructura visual movil corresponde a la Fase 12.

```ts
export type PersonalAreaSection = {
  description: string;
  id: PersonalAreaSectionId;
  label: string;
  path: string;
};

const SECTIONS: Record<PersonalAreaSectionId, PersonalAreaSection> = {
  datos: {
    description:
      "Consulte y actualice los datos de su cuenta que SIPEG le permite cambiar. El correo, la cedula y el rol los administra SIPEG.",
    id: "datos",
    label: "Datos de la cuenta",

    id: "seguridad",
    label: "Seguridad de la cuenta",
    path: "/perfil/seguridad",
  },
  actividades: {
    description:
      "Consulte sus inscripciones, el estado de su asistencia y el codigo de acceso de cada actividad.",
    id: "actividades",
    label: "Mis actividades",
    path: "/perfil/actividades",
  },
  certificados: {
    description:
      "Consulte y descargue los certificados de las actividades en las que su asistencia fue confirmada.",
    id: "certificados",
    label: "Mis certificados",
    path: "/perfil/certificados",
  },
};

/** Orden de navegacion del submenu. El orden de claves de `SECTIONS` es el orden de la interfaz. */
export const PERSONAL_AREA_SECTIONS: readonly PersonalAreaSection[] = Object.values(SECTIONS);

const DEFAULT_SECTION_ID: PersonalAreaSectionId = "datos";

/**
 * Resuelve la seccion activa a partir de la ruta. Una ruta desconocida, incluido el alias `/perfil`,
 * cae en la seccion de datos, de modo que el submenu nunca queda sin seccion marcada.
 */
export function resolvePersonalAreaSection(pathname: string): PersonalAreaSection {
  return (
    PERSONAL_AREA_SECTIONS.find((section) => section.path === pathname) ??
    SECTIONS[DEFAULT_SECTION_ID]
  );
}
```

```tsx
import { Box, Button, Flex, Heading, Stack, Text } from "@chakra-ui/react";
import type { KeyboardEvent } from "react";
import { NavLink } from "react-router";

import { Surface } from "@/components";

import { PERSONAL_AREA_SECTIONS, type PersonalAreaSectionId } from "../model/personalAreaSections";

export const PERSONAL_AREA_NAV_ID = "personal-area-sections";

export type PersonalAreaNavigationProps = {
  activeSectionId: PersonalAreaSectionId;
  isDesktop: boolean;
  isOpen: boolean;
  onKeyDown: (event: KeyboardEvent<HTMLDivElement>) => void;
  onOpenChange: (isOpen: boolean) => void;
  onSelect: (path: string) => void;
  toggleRef?: React.Ref<HTMLButtonElement>;
};

function activeSectionLabel(activeSectionId: PersonalAreaSectionId) {
  return PERSONAL_AREA_SECTIONS.find((section) => section.id === activeSectionId)?.label ?? "";
}

/**
 * Submenu of the personal area. Desktop keeps a persistent side navigation; mobile replaces it with a
 * local disclosure that keeps the current section name visible while collapsed and lists the four
 * sections vertically when opened. Visibility is decided by the layout, not by a media query, so the
 * open state is observable in tests. `aria-current` and the filled versus outlined shape both mark
 * the active section, so color is never the only cue.
 */
export function PersonalAreaNavigation({
  activeSectionId,
  isDesktop,
  isOpen,
  onKeyDown,
  onOpenChange,
  onSelect,
  toggleRef,
}: PersonalAreaNavigationProps) {
  const isVisible = isDesktop || isOpen;

  return (
    <Stack gap={3} onKeyDown={onKeyDown}>
      {isDesktop ? null : (
        <Surface padding="normal">
          <Stack gap={3}>
            <Box>
              <Text
                color="text.muted"
                fontSize="sm"
                fontWeight="800"
                letterSpacing="0.1em"
                textTransform="uppercase"
              >
                Seccion actual
              </Text>
              <Heading as="p" color="text.default" fontFamily="heading" fontSize="xl" mt={1}>
                {activeSectionLabel(activeSectionId)}
              </Heading>
            </Box>
            <Button
              aria-controls={PERSONAL_AREA_NAV_ID}
              aria-expanded={isOpen}
              colorPalette="terracotta"
              minH="44px"
              onClick={() => onOpenChange(!isOpen)}
              ref={toggleRef}
              rounded="full"
              variant="outline"
              w="full"
            >
              {isOpen ? "Ocultar secciones" : "Ver todas las secciones"}
            </Button>
          </Stack>
        </Surface>
      )}

      <Box
        as="nav"
        aria-label="Secciones del area personal"
        display={isVisible ? "block" : "none"}
        id={PERSONAL_AREA_NAV_ID}
      >
        <Flex as="ul" direction="column" gap={2} listStyle="none" m={0} p={0} w="full">
          {PERSONAL_AREA_SECTIONS.map((section) => (
            <Box as="li" key={section.id} w="full">
              <NavLink
                end
                onClick={() => onSelect(section.path)}
                style={{ display: "block", textDecoration: "none" }}
                to={section.path}
              >
                {({ isActive }) => (
                  <Box
                    as="span"
                    bg={isActive ? "accent.solid" : "surface.raised"}
                    borderColor={isActive ? "accent.solid" : "border.subtle"}
                    borderWidth="1px"
                    color={isActive ? "accent.contrast" : "text.default"}
                    display="flex"
                    fontSize="sm"
                    fontWeight="800"
                    itemsCenter
                    minH="44px"
                    px={4}
                    py={2.5}
                    rounded="xl"
                  >
                    {section.label}
                  </Box>
                )}
              </NavLink>
            </Box>
          ))}
        </Flex>
      </Box>
    </Stack>
  );
}
```

```tsx
import { Box, Flex, useMediaQuery } from "@chakra-ui/react";
import { useCallback, useEffect, useRef, useState, type KeyboardEvent } from "react";
import { Outlet, useLocation } from "react-router";

import { ModuleShell } from "@/components";

import { resolvePersonalAreaSection } from "../model/personalAreaSections";

import { PersonalAreaNavigation } from "./PersonalAreaNavigation";

const DESKTOP_VIEWPORT_QUERY = "(min-width: 48em)";

/**
 * Common frame of the personal area. It owns the single `h1` of the area, the submenu and the
 * `Outlet` of the active section, so every section has its own route while keeping the same
 * navigation. The submenu visibility comes from `useMediaQuery` instead of a media query, so the
 * expanded state is a real state value that tests can observe and `Escape` can act on. Choosing a
 * section closes the mobile disclosure and moves focus to the destination heading, because a screen
 * reader user would otherwise stay inside a menu that closed underneath them.
 */
export function PersonalAreaLayout() {
  const location = useLocation();
  const headingRef = useRef<HTMLHeadingElement>(null);
  const toggleRef = useRef<HTMLButtonElement>(null);
  const [isDesktop] = useMediaQuery([DESKTOP_VIEWPORT_QUERY]);
  const [isMenuOpen, setIsMenuOpen] = useState(false);
  const shouldFocusHeading = useRef(false);
  const activeSection = resolvePersonalAreaSection(location.pathname);

  useEffect(() => {
    if (!shouldFocusHeading.current) {
      return;
    }

    shouldFocusHeading.current = false;
    setIsMenuOpen(false);
    headingRef.current?.focus();
  }, [location.pathname]);

  const handleSelect = useCallback(
    (path: string) => {
      setIsMenuOpen(false);

      if (path === location.pathname) {
        headingRef.current?.focus();
        return;
      }

      shouldFocusHeading.current = true;
    },
    [location.pathname],
  );

  const handleKeyDown = useCallback(
    (event: KeyboardEvent<HTMLDivElement>) => {
      if (isDesktop || !isMenuOpen || event.key !== "Escape") {
        return;
      }

      setIsMenuOpen(false);
      toggleRef.current?.focus();
    },
    [isDesktop, isMenuOpen],
  );

  return (
    <Box maxW="5xl" mx="auto" py={{ base: 3, md: 6 }}>
      <ModuleShell
        description="Consulte y mantenga los datos de su cuenta, su seguridad y los servicios asociados a su participacion."
        headingLabel="Mi perfil"
        headingRef={headingRef}
        title="Area personal"
      >
        <Flex align="start" direction={{ base: "column", md: "row" }} gap={{ base: 6, md: 8 }}>
          <Box flex="0 0 auto" w={{ base: "full", md: "17rem" }}>
            <PersonalAreaNavigation
              activeSectionId={activeSection.id}
              isDesktop={isDesktop}
              isOpen={isMenuOpen}
              onKeyDown={handleKeyDown}
              onOpenChange={setIsMenuOpen}
              onSelect={handleSelect}
              toggleRef={toggleRef}
            />
          </Box>
          <Box flex="1 1 auto" minW={0} w="full">
            <Outlet />
          </Box>
        </Flex>
      </ModuleShell>
    </Box>
  );
}
```
